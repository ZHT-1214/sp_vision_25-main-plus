cmake_minimum_required(VERSION 3.22.0)
set(lib_name foxglove)

set(foxglove_SOURCE_DIR ${CMAKE_CURRENT_SOURCE_DIR})

file(GLOB foxglove_lib_source CONFIGURE_DEPENDS
    "${foxglove_SOURCE_DIR}/src/*.cpp"
    "${foxglove_SOURCE_DIR}/src/server/*.cpp"
)
add_library(${lib_name}_lib SHARED ${foxglove_lib_source})
target_include_directories(${lib_name}_lib PUBLIC ${foxglove_SOURCE_DIR}/include)
target_link_libraries(${lib_name}_lib PRIVATE ${foxglove_SOURCE_DIR}/lib/libfoxglove.a)


add_library(${lib_name}_interface_lib INTERFACE)
target_include_directories(${lib_name}_interface_lib INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/autogenerated_flatbuffers)
